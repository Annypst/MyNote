 （3）ProdDao.java

```java
public class ProdDao {
    //find by name method
    public Product findByName(String name) throws Exception {
        Connection con=JDBCUtils.getConnection();
        String sql="SELECT * FROM products WHERE name= ?";
        PreparedStatement pre=con.prepareStatement(sql);
        pre.setString(1,name);
        ResultSet res=pre.executeQuery();
        Product pro=new Product();
        if(res.next()){
            pro.setName(res.getString("name"));
            pro.setPrice(res.getString("price"));
        }
        pre.close();
        con.close();
        return pro;
    }
    //insert method
    public int insert(Product prod) throws Exception{
        Connection con=JDBCUtils.getConnection();
        String sql="INSERT INTO products(name,price) VALUES(?,?)";
        PreparedStatement pre=con.prepareStatement(sql);
        pre.setString(1,prod.getName());
        pre.setString(2,prod.getPrice());
        int num=pre.executeUpdate();//返回受影响行数
        pre.close();
        con.close();
        return num;
    }
}

```

# 3.Controller

ProdServlet

```java
@WebServlet(name="ProdServlet",urlPatterns = "/exam/ProdServlet")
public class ProdServlet extends HttpServlet {
    protected void doPost(HttpServletRequest req, HttpServletResponse resp) throws ServletException, IOException{
        req.setCharacterEncoding("utf-8");
        String name=req.getParameter("pname");
        String price=req.getParameter("pprice");
        ProdDao dao=new ProdDao();
        Product pro=new Product();

        if(dao.findByName(name).getName()!=null){
            resp.sendRedirect("/exam/addProd.jsp"); //已存在
        }
        else {
            pro.setName(name);
            pro.setPrice(price);
            try {
                int num = dao.insert(pro); // 调用可能引发异常的方法

                if (num > 0) {
                    // 影响行数大于 0，说明数据库写入成功，执行成功跳转
                    resp.sendRedirect("/exam/addSuccess.jsp");
                } else {
                    // 虽未触发数据库报错，但影响行数为 0，视为插入失败
                    resp.sendRedirect("/exam/addProd.jsp");
                }
            } catch (Exception e) {
                e.printStackTrace();
                resp.sendRedirect("/exam/addProd.jsp"); // 发生异常一律视为插入失败
            }
        }
        }
    }

```
